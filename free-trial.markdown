// Azure Function URL for trial signup
const AZURE_FUNCTION_URL = 'https://whisperscape-sms-form-response.azurewebsites.net/api/UrlProvisionWebhook';

document.getElementById('trialSignupForm').addEventListener('submit', async function(e) {
  e.preventDefault();
  
  const submitBtn = document.getElementById('submitBtn');
  const form = document.getElementById('trialSignupForm');
  const successDiv = document.getElementById('formSuccess');
  const errorDiv = document.getElementById('formError');
  
  // Show loading state
  submitBtn.classList.add('loading');
  submitBtn.disabled = true;
  
  // Gather form data - strip all spaces from phone number
  const phoneRaw = document.getElementById('phone').value.trim();
  const phoneClean = phoneRaw.replace(/\s/g, ''); // Remove all spaces
  
  const formData = {
    'Full Name': document.getElementById('fullName').value.trim(),
    'Email Address': document.getElementById('email').value.trim(),
    'Phone Number': phoneClean,
    'Channel': document.querySelector('input[name="Channel"]:checked').value,
    'trialType': 'free'
  };
  
  try {
    const response = await fetch(AZURE_FUNCTION_URL, {
      method: 'POST',
      mode: 'cors',
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json'
      },
      body: JSON.stringify(formData)
    });
    
    // Check if response is OK before parsing
    if (!response.ok) {
      const errorText = await response.text();
      console.error('Server error response:', errorText);
      throw new Error(`Server error: ${response.status} - ${errorText || 'Unknown error'}`);
    }
    
    const result = await response.json();
    
    if (response.ok && result.success) {
      // Success!
      form.style.display = 'none';
      successDiv.style.display = 'block';
      
      // Populate success details
      const streamUrl = result.streamUrl || `https://listen2.ejukebox.net/${result.urlSuffix}`;
      document.querySelector('.success-message').textContent = result.existingTrial 
        ? `Great news! You already have an active ${result.channel} trial.`
        : `Your ${result.channel} music stream is now ready!`;
      
      document.querySelectorAll('.stream-url').forEach(el => {
        el.href = streamUrl;
        el.textContent = streamUrl;
      });
      document.querySelectorAll('.stream-url-btn').forEach(el => {
        el.href = streamUrl;
      });
      document.querySelector('.expiry-date').textContent = result.expiryDateFormatted || new Date(result.expiryDate).toLocaleDateString('en-AU');
      
    } else {
      // Error from server
      throw new Error(result.error || 'Failed to create trial. Please try again.');
    }
    
  } catch (error) {
    console.error('Form submission error:', error);
    form.style.display = 'none';
    errorDiv.style.display = 'block';
    document.querySelector('.error-message').textContent = error.message || 'An unexpected error occurred. Please try again.';
  } finally {
    submitBtn.classList.remove('loading');
    submitBtn.disabled = false;
  }
});

function resetForm() {
  const form = document.getElementById('trialSignupForm');
  const successDiv = document.getElementById('formSuccess');
  const errorDiv = document.getElementById('formError');
  
  form.style.display = 'block';
  successDiv.style.display = 'none';
  errorDiv.style.display = 'none';
  form.reset();
}

// Phone number formatting
document.getElementById('phone').addEventListener('input', function(e) {
  let value = e.target.value.replace(/\D/g, '');
  
  // Handle various formats
  if (value.startsWith('61')) {
    value = '0' + value.substring(2);
  }
  
  // Format as 04XX XXX XXX
  if (value.length > 4 && value.length <= 7) {
    value = value.substring(0, 4) + ' ' + value.substring(4);
  } else if (value.length > 7) {
    value = value.substring(0, 4) + ' ' + value.substring(4, 7) + ' ' + value.substring(7, 10);
  }
  
  e.target.value = value;
});
</script>
